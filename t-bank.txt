const STORAGE_KEY = staticgram_tbank_demo_v1;

let state = loadState();

function loadState() {
    try {
        const saved = localStorage.getItem(STORAGE_KEY);

        if (saved) {
            const parsed = JSON.parse(saved);

            if (
                Number.isFinite(parsed.stars) &&
                Number.isFinite(parsed.gram) &&
                Array.isArray(parsed.transactions)
            ) {
                return parsed;
            }
        }
    } catch (e) {}

    return JSON.parse(JSON.stringify(initialState));
}

function saveState() {
    try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
    } catch (e) {
        showToast(Не удалось сохранить данные браузера);
    }
}

function formatNumber(number) {
    return Math.floor(number).toLocaleString(ru-RU);
}

function currencyName(currency) {
    return currency === stars  Stars  Gram;
}

function currencyIcon(currency) {
    return currency === stars  ★  ◇;
}

function moneyText(amount, currency) {
    const sign = amount  0  +  amount  0  −  ;
    return sign + formatNumber(Math.abs(amount)) +   +
        currencyName(currency);
}

function renderTransactions(containerId, limit = Infinity) {
    const container = document.getElementById(containerId);

    const items = state.transactions.slice(0, limit);

    if (!items.length) {
        container.innerHTML = `
            div class=empty
                Операций пока нет.br
                Здесь появится история вашего счёта.
            div
        `;
        return;
    }

    container.innerHTML = items.map(tx = {
        const positive = tx.amount  0;

        return `
            div class=transaction
                div class=transaction-icon
                    ${escapeHtml(tx.icon  currencyIcon(tx.currency))}
                div

                div class=transaction-info
                    div class=transaction-name
                        ${escapeHtml(tx.title)}
                    div
                    div class=transaction-date
                        ${escapeHtml(tx.date)} · Демо
                    div
                div

                div class=transaction-amount ${positive  positive  negative}
                    ${escapeHtml(moneyText(tx.amount, tx.currency))}
                div
            div
        `;
    }).join();
}

function escapeHtml(value) {
    return String(value).replace([&']g, character = ({
        & &amp;,
         &lt;,
         &gt;,
        '' &quot;,
        ' &#039;
    })[character]);
}

function render() {
    document.getElementById(homeStars).textContent =
        formatNumber(state.stars) +  Stars;

    document.getElementById(homeGram).textContent =
        formatNumber(state.gram) +  Gram;

    document.getElementById(starsDescription).textContent =
        formatNumber(state.stars) +  Stars;

    document.getElementById(gramDescription).textContent =
        formatNumber(state.gram) +  Gram;

    document.getElementById(profileStars).textContent =
        formatNumber(state.stars) +  Stars;

    document.getElementById(profileGram).textContent =
        formatNumber(state.gram) +  Gram;

    renderTransactions(homeTransactions, 3);
    renderTransactions(allTransactions);

    saveState();

    updateExchangeHint();
}

function addTransaction(title, amount, currency, icon) {
    const now = new Date();

    state.transactions.unshift({
        title,
        amount,
        currency,
        icon,
        date now.toLocaleDateString(ru-RU) + ,  +
            now.toLocaleTimeString(ru-RU, {
                hour 2-digit,
                minute 2-digit
            })
    });

    state.transactions = state.transactions.slice(0, 100);
}

function openPage(page) {
    const target = document.getElementById(page);

    if (!target) return;

    document.querySelectorAll(.page).forEach(element = {
        element.classList.remove(active);
    });

    target.classList.add(active);

    document.querySelectorAll(.nav-button).forEach(button = {
        button.classList.toggle(
            active,
            button.dataset.page === page
        );
    });

    window.scrollTo({ top 0, behavior smooth });
}

let toastTimer;

function showToast(message) {
    const element = document.getElementById(toast);

    element.textContent = message;
